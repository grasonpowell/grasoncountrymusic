const express = require('express'); const mongoose = require('mongoose'); const bcrypt = require('bcryptjs'); const jwt = require('jwt-simple'); const nodemailer = require('nodemailer'); const cors = require('cors'); const app = express(); app.use(express.json()); app.use(cors({ origin: 'https://grasoncountrymusic.com', credentials: true })); const JWT_SECRET = process.env.JWT_SECRET || 'super-secret-key-grason-country'; const ADMIN_EMAIL = 'grasoncountrymusic@gmail.com'; // DB Schemas const UserSchema = new mongoose.Schema({ name: String, email: { type: String, unique: true, required: true }, passwordHash: String, isArtist: { type: Boolean, default: false }, twoFactorCode: String, twoFactorExpires: Date }); const PostSchema = new mongoose.Schema({ authorName: String, authorEmail: String, isArtist: Boolean, caption: String, mediaUrl: String, mediaType: String, firewood: { type: Number, default: 0 }, likes: [String], comments: [{ author: String, text: String, createdAt: { type: Date, default: Date.now } }], createdAt: { type: Date, default: Date.now } }); const PollSchema = new mongoose.Schema({ question: String, options: [{ text: String, votes: { type: Number, default: 0 } }], active: { type: Boolean, default: true } }); const User = mongoose.model('User', UserSchema); const Post = mongoose.model('Post', PostSchema); const Poll = mongoose.model('Poll', PollSchema); // Email Transporter for 2FA const transporter = nodemailer.createTransport({ service: 'gmail', auth: { user: process.env.EMAIL_USER, pass: process.env.EMAIL_PASS } }); // Middleware: Authenticate Token const requireAuth = (req, res, next) => { const token = req.headers.authorization?.split(' ')[1]; if (!token) return res.status(401).json({ error: 'Unauthorized access' }); try { req.user = jwt.decode(token, JWT_SECRET); next(); } catch (err) { res.status(401).json({ error: 'Invalid token' }); } }; // Middleware: Require Artist Admin const requireArtist = (req, res, next) => { if (!req.user || !req.user.isArtist) { return res.status(403).json({ error: 'Forbidden: Artist privileges required' }); } next(); }; /* --- AUTHENTICATION & 2FA --- */ app.post('/api/auth/login', async (req, res) => { const { email, password } = req.body; const user = await User.findOne({ email: email.toLowerCase() }); if (!user || !(await bcrypt.compare(password, user.passwordHash))) { return res.status(400).json({ error: 'Invalid credentials' }); } // Trigger 2FA for Artist Account if (user.email === ADMIN_EMAIL) { const code = Math.floor(100000 + Math.random() * 900000).toString(); user.twoFactorCode = code; user.twoFactorExpires = Date.now() + 10 * 60 * 1000; // 10 minutes await user.save(); await transporter.sendMail({ from: '"Grason Powell Security" ', to: ADMIN_EMAIL, subject: '🔑 Your Artist Security Verification Code', text: `Your 2FA login verification code is: ${code}` }); return res.json({ require2FA: true, email: user.email }); } const token = jwt.encode({ id: user._id, name: user.name, email: user.email, isArtist: user.isArtist }, JWT_SECRET); res.json({ token, user: { name: user.name, email: user.email, isArtist: user.isArtist } }); }); app.post('/api/auth/verify-2fa', async (req, res) => { const { email, code } = req.body; const user = await User.findOne({ email: email.toLowerCase() }); if (!user || user.twoFactorCode !== code || Date.now() > user.twoFactorExpires) { return res.status(400).json({ error: 'Invalid or expired 2FA verification code' }); } user.twoFactorCode = null; user.twoFactorExpires = null; await user.save(); const token = jwt.encode({ id: user._id, name: user.name, email: user.email, isArtist: true }, JWT_SECRET); res.json({ token, user: { name: user.name, email: user.email, isArtist: true } }); }); /* --- DYNAMIC POLL ENGINE --- */ app.get('/api/poll', async (req, res) => { const poll = await Poll.findOne({ active: true }); res.json(poll); }); app.post('/api/poll/update', requireAuth, requireArtist, async (req, res) => { const { question, options } = req.body; // options is an array of string items await Poll.updateMany({}, { active: false }); const newPoll = new Poll({ question, options: options.map(opt => ({ text: opt, votes: 0 })), active: true }); await newPoll.save(); res.json({ success: true, poll: newPoll }); }); app.post('/api/poll/vote', async (req, res) => { const { pollId, optionIndex } = req.body; const poll = await Poll.findById(pollId); if (poll && poll.options[optionIndex]) { poll.options[optionIndex].votes += 1; await poll.save(); } res.json(poll); }); /* --- SOCIAL MEDIA FEED API --- */ app.get('/api/posts', async (req, res) => { const posts = await Post.find().sort({ createdAt: -1 }); res.json(posts); }); app.post('/api/posts', requireAuth, async (req, res) => { const { caption, mediaUrl, mediaType } = req.body; const post = new Post({ authorName: req.user.name, authorEmail: req.user.email, isArtist: req.user.isArtist, caption, mediaUrl, mediaType }); await post.save(); res.json(post); }); app.post('/api/posts/:id/comment', requireAuth, async (req, res) => { const post = await Post.findById(req.params.id); post.comments.push({ author: req.user.name, text: req.body.text }); await post.save(); res.json(post); }); app.post('/api/posts/:id/firewood', async (req, res) => { const post = await Post.findById(req.params.id); post.firewood += 1; await post.save(); res.json(post); }); app.listen(5000, () => console.log('🔒 Security Server Running on Port 5000'));
